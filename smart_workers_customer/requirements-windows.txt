# ── 1. System runtime ────────────────────────────────────────────────────────
# Easiest path is via winget. Run an elevated PowerShell:
#
#   winget install --id Microsoft.OpenJDK.17       --silent
#   winget install --id Git.Git                    --silent
#   winget install --id GitHub.cli                 --silent     # optional
#   winget install --id stedolan.jq                --silent     # for release.sh
#   winget install --id Google.AndroidStudio       --silent
#
# After install, open a new shell so PATH refreshes. Confirm:
#   java -version    → 17.x
#   git --version

openjdk==17                       # JDK 17 — required by AGP
git==latest
jq>=1.6                           # parsed by release.sh

# ── 2. Flutter SDK ───────────────────────────────────────────────────────────
# Download stable from https://docs.flutter.dev/get-started/install/windows
# and extract to C:\src\flutter   (avoid Program Files due to spaces).
# Add C:\src\flutter\bin to your PATH (System Properties → Environment Variables).
# Verify in a new PowerShell:
#   flutter --version
#   flutter doctor

flutter-sdk>=3.24.0,<4.0.0

# ── 3. Android toolchain ─────────────────────────────────────────────────────
# In Android Studio: SDK Manager → install Platform 34 / 35,
# Build-Tools 34.0.0+, NDK 26.x, command-line tools.
# Then in a shell:
#   flutter doctor --android-licenses    (accept all)
#
# Bash isn't required to BUILD, but `release.sh` is a bash script. To run it
# on Windows you have two choices:
#   (a) Use Git Bash (comes with Git for Windows) — easiest.
#   (b) Use WSL (Ubuntu) — see requirements-linux.txt inside WSL instead.

# ── 6. Release keystore (one-time, Windows path) ────────────────────────────
# In PowerShell:
#   keytool -genkey -v `
#     -keystore "$env:USERPROFILE\upload-keystore.jks" `
#     -alias smartworkers `
#     -keyalg RSA -keysize 2048 -validity 10000
#
# Then:
#   copy android\key.properties.example android\key.properties
# Edit android\key.properties; storeFile MUST be the full Windows path with
# DOUBLE backslashes or forward slashes, e.g.:
#     storeFile=C:/Users/you/upload-keystore.jks
#
# Extract SHA-1 / SHA-256:
#   keytool -list -v `
#     -keystore "$env:USERPROFILE\upload-keystore.jks" `
#     -alias smartworkers
# Register both fingerprints in Firebase Console; re-download
# google-services.json. (See FIREBASE_OTP_FIX.md.)
